# api-doc.html Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build `api-doc.html`, a single-file tool that pastes a JSDoc comment from the clipboard and renders it as a card-based HTML view, with @param/@returns/@link color-coded and errors surfaced in a red banner.

**Architecture:** Pure parser (`parse-jsdoc.js`) separated from DOM glue, so the parser is unit-testable with Node's built-in test runner. HTML file loads the parser via `<script src>` and contains all CSS + render + clipboard glue. The renderer is manual-tested in the browser (visual output).

**Tech Stack:** Vanilla HTML/CSS/JS. `navigator.clipboard.readText()`. Node 18+ built-in `node:test` for parser tests.

---

## File Structure

- **Create:** `parse-jsdoc.js` — pure parser (`parseJsDoc` + `InvalidFormatError`), no DOM access, dual-loadable in Node (`module.exports`) and browser (assigns to `window.ParseJsDoc`).
- **Create:** `tests/parse-jsdoc.test.js` — Node test runner suite for the parser.
- **Modify:** `api-doc.html` — loads `parse-jsdoc.js`, contains CSS theme + `renderToDOM()` + `showError()` + `onPaste()` glue. Replaces the existing 0-byte stub.

No other files in the repo are touched. Existing tools (`code-explain.html` etc.) are not modified.

Replace `parse-jsdoc.js` with the following single-pass version. Tag lines dispatch at the top; body lines flow into the active section. Tasks 4 and 6 will extend the `'param'` and `'link'` body handlers.

```js
class InvalidFormatError extends Error {
  constructor(message) {
    super(message);
    this.name = 'InvalidFormatError';
  }
}

const PREFIX_PATTERN = /^\s*\*{1,2}\s?/;
const PARAM_HEAD = /^@param\s+(\S+)\s*(.*)$/;

function stripPrefix(line) {
  return line.replace(PREFIX_PATTERN, '');
}

function splitBody(text) {
  const inner = text.replace(/^\s*\/\*\*/, '').replace(/\*\/\s*$/, '');
  return inner.split('\n').map(stripPrefix);
}

function parseJsDoc(text) {
  const trimmed = (text ?? '').trim();
  if (!trimmed.startsWith('/**') || !trimmed.endsWith('*/')) {
    throw new InvalidFormatError('Text is not a JSDoc comment');
  }
  const lines = splitBody(trimmed);
  const data = {
    description: [],
    params: [],
    returns: null,
    links: []
  };
  let section = 'desc'; // 'desc' | 'param' | 'example' | 'returns' | 'link' | 'unknown'
  let currentParam = null;

  for (const line of lines) {
    // Tag dispatch: each tag may switch section
    if (line.startsWith('@param ')) {
      const head = line.match(PARAM_HEAD);
      currentParam = { name: head[1], desc: head[2], example: null };
      data.params.push(currentParam);
      section = 'param';
      continue;
    }
    if (line.startsWith('@returns ')) {
      data.returns = line.slice('@returns '.length);
      section = 'returns';
      currentParam = null;
      continue;
    }
    if (line.startsWith('@link')) {
      section = 'link';
      currentParam = null;
      continue;
    }
    if (line.startsWith('@')) {
      const tag = line.match(/^\S+/)[0];
      console.warn(`parseJsDoc: skipping unknown tag ${tag}`);
      section = 'unknown';
      currentParam = null;
      continue;
    }

    // Body lines per active section
    if (section === 'param') {
      // (Task 4 will add the 例如: detection branch here)
      currentParam.desc = currentParam.desc ? `${currentParam.desc}\n${line}` : line;
      continue;
    }
    if (section === 'returns') {
      data.returns = data.returns ? `${data.returns}\n${line}` : line;
      continue;
    }
    if (section === 'link') {
      // (Task 6 will populate data.links here)
      continue;
    }
    if (section === 'unknown') {
      continue; // skip body until next @
    }
    // section === 'desc' (default)
    data.description.push(line);
  }
  return data;
}

if (typeof module !== 'undefined' && module.exports) {
  module.exports = { parseJsDoc, InvalidFormatError };
}
if (typeof window !== 'undefined') {
  window.ParseJsDoc = { parseJsDoc, InvalidFormatError };
}
```

    <script>
        const { parseJsDoc, InvalidFormatError } = window.ParseJsDoc;

        function humanizeError(err) {
            if (err instanceof InvalidFormatError) {
                return '剪贴板内容不是有效的 JSDoc 注释(需以 /** 开头、*/ 结尾)。';
            }
            if (err && err.name === 'NotAllowedError') {
                return '无法读取剪贴板,请检查浏览器权限。';
            }
            return `读取失败: ${err && err.message ? err.message : String(err)}`;
        }

        function showError(message) {
            const el = document.getElementById('error');
            if (message == null) {
                el.style.display = 'none';
                el.textContent = '';
            } else {
                el.textContent = message;
                el.style.display = 'block';
            }
        }

        function renderToDOM(data) {
            const out = document.getElementById('output');
            out.innerHTML = '';

            if (data.description.length) {
                const card = document.createElement('div');
                card.className = 'card desc';
                data.description.forEach((line, i) => {
                    const p = document.createElement('p');
                    p.className = 'desc-body';
                    p.textContent = line || ' ';
                    if (i > 0) p.style.marginTop = '8px';
                    card.appendChild(p);
                });
                out.appendChild(card);
            }

            for (const param of data.params) {
                const card = document.createElement('div');
                card.className = 'card param';
                const tag = document.createElement('span');
                tag.className = 'tag';
                tag.textContent = '@param';
                const name = document.createElement('span');
                name.className = 'name';
                name.textContent = param.name;
                card.appendChild(tag);
                card.appendChild(name);
                const body = document.createElement('p');
                body.className = 'desc-body';
                body.textContent = param.desc;
                card.appendChild(body);
                if (param.example != null) {
                    const ex = document.createElement('div');
                    ex.className = 'example';
                    const label = document.createElement('div');
                    label.className = 'example-label';
                    label.textContent = '例如';
                    const pre = document.createElement('pre');
                    pre.textContent = param.example;
                    ex.appendChild(label);
                    ex.appendChild(pre);
                    card.appendChild(ex);
                }
                out.appendChild(card);
            }

            if (data.returns != null) {
                const card = document.createElement('div');
                card.className = 'card returns';
                const tag = document.createElement('span');
                tag.className = 'tag';
                tag.textContent = '@returns';
                const body = document.createElement('p');
                body.className = 'desc-body';
                body.textContent = data.returns;
                card.appendChild(tag);
                card.appendChild(body);
                out.appendChild(card);
            }

            if (data.links.length) {
                const card = document.createElement('div');
                card.className = 'card link';
                const tag = document.createElement('span');
                tag.className = 'tag';
                tag.textContent = '@link';
                card.appendChild(tag);
                const ol = document.createElement('ol');
                for (const link of data.links) {
                    const li = document.createElement('li');
                    const strong = document.createElement('strong');
                    strong.textContent = link.name;
                    li.appendChild(strong);
                    li.appendChild(document.createTextNode(`: ${link.desc}`));
                    ol.appendChild(li);
                }
                card.appendChild(ol);
                out.appendChild(card);
            }
        }

        async function onPaste() {
            try {
                const text = await navigator.clipboard.readText();
                const data = parseJsDoc(text);
                renderToDOM(data);
                showError(null);
            } catch (err) {
                showError(humanizeError(err));
            }
        }

        document.getElementById('btn-paste').addEventListener('click', onPaste);
    </script>
</body>
</html>
