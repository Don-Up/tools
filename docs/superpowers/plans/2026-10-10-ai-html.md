# ai.html Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a single-file `ai.html` Q&A page that streams answers from MiniMax M3 (China endpoint) with optional thinking-block display and Markdown-rendered output.

**Architecture:** Single self-contained HTML file. Vanilla JS reads `localStorage` for API key + toggle, builds an Anthropic-compatible SSE request, parses the stream with `fetch` + `ReadableStream`, accumulates `text` and `thinking` deltas in separate buffers, and renders the final answer via marked.js. CSS follows the project's existing dark theme (`#1e1e1e` bg, `#4fc3f7` accent).

**Tech Stack:** Plain HTML/CSS/JS, `marked@12.0.0` from jsDelivr CDN, native `fetch` + `TextDecoder` for SSE, no build step.

**Reference spec:** `docs/superpowers/specs/2026-10-10-ai-html-design.md`

---

## File Structure

Only one file is created — `ai.html` in the project root. The plan builds it incrementally, layer by layer:

| Layer | Responsibility |
| --- | --- |
| `<style>` block | Dark theme, layout, component styles |
| `<body>` skeleton | Static DOM: top bar, settings row, error banner, thinking section, answer section, input bar |
| State + utils | `localStorage` keys, helpers (escape, debounce) |
| API key UI | Mask/edit/save flow |
| `send()` + SSE parser | Build request, stream events, route deltas to buffers |
| Renderers | Thinking plain-text, text cursor, final marked render, code-block copy buttons |
| Error handling | Status code mapping, AbortError suppression, banner show/hide |
| Hotkeys + UX | Ctrl+Enter, Esc, auto-grow textarea, maxlength flash |

Each task appends to a single function/section so the engineer can verify in the browser between tasks.

```javascript
function buildRequestBody(question) {
    const body = {
        model: MODEL,
        max_tokens: MAX_TOKENS,
        system: SYSTEM_PROMPT,
        messages: [{ role: 'user', content: question }],
        stream: true,
    };
    if (state.showThinking) {
        body.thinking = { type: 'enabled' };
    }
    return body;
}

async function callApi(question) {
    const body = buildRequestBody(question);
    const res = await fetch(`${BASE_URL}/v1/messages`, {
        method: 'POST',
        headers: {
            'content-type': 'application/json',
            'x-api-key': state.apiKey,
            'anthropic-version': '2023-06-01',
            'anthropic-dangerous-direct-browser-access': 'true',
        },
        body: JSON.stringify(body),
        signal: state.abortController.signal,
    });

    if (!res.ok) {
        const err = new Error(`HTTP ${res.status}`);
        err.status = res.status;
        throw err;
    }

    const contentType = res.headers.get('content-type') || '';
    if (!contentType.includes('text/event-stream')) {
        const json = await res.json();
        return handleNonStreaming(json);
    }

    await parseSseStream(res.body);
}

async function handleNonStreaming(json) {
    if (Array.isArray(json && json.content)) {
        for (const block of json.content) {
            if (block.type === 'text') state.textBuf += block.text;
            else if (block.type === 'thinking') state.thinkingBuf += (block.thinking || '');
        }
    }
    state.stopReason = json.stop_reason || 'end_turn';
}

async function parseSseStream(body) {
    const reader = body.getReader();
    const decoder = new TextDecoder('utf-8');
    let buffer = '';
    while (true) {
        const { value, done } = await reader.read();
        if (done) break;
        buffer += decoder.decode(value, { stream: true });
        let sep;
        while ((sep = buffer.indexOf('\n\n')) !== -1) {
            const raw = buffer.slice(0, sep);
            buffer = buffer.slice(sep + 2);
            processSseEvent(raw);
        }
    }
    if (buffer.trim()) processSseEvent(buffer);
}

function processSseEvent(raw) {
    const lines = raw.split('\n');
    let eventName = '';
    const dataLines = [];
    for (const line of lines) {
        if (line.startsWith('event:')) eventName = line.slice(6).trim();
        else if (line.startsWith('data:')) dataLines.push(line.slice(5).trim());
    }
    const data = dataLines.join('\n');
    if (!data || data === '[DONE]') return;
    let payload;
    try {
        payload = JSON.parse(data);
    } catch (err) {
        console.warn('SSE JSON parse failed', err, data);
        return;
    }
    handleSsePayload(eventName, payload);
}

function handleSsePayload(eventName, payload) {
    switch (eventName) {
        case 'message_start':
            if (payload.message && payload.message.stop_reason) {
                state.stopReason = payload.message.stop_reason;
            }
            break;
        case 'content_block_start':
            state.currentBlockIndex = payload.index;
            state.currentBlockType = payload.content_block && payload.content_block.type;
            break;
        case 'content_block_delta': {
            const idx = payload.index;
            if (idx !== state.currentBlockIndex) {
                state.currentBlockIndex = idx;
            }
            const delta = payload.delta || {};
            if (delta.type === 'text_delta' || typeof delta.text === 'string') {
                state.textBuf += delta.text;
                renderStreamingText();
            } else if (delta.type === 'thinking_delta' || typeof delta.thinking === 'string') {
                state.thinkingBuf += delta.thinking;
                renderStreamingThinking();
            }
            break;
        }
        case 'content_block_stop':
            if (state.currentBlockType === 'text') {
                renderFinalAnswer();
            }
            state.currentBlockType = null;
            break;
        case 'message_delta':
            if (payload.delta && payload.delta.stop_reason) {
                state.stopReason = payload.delta.stop_reason;
            }
            break;
        case 'message_stop':
            break;
        default:
            break;
    }
}

    if (c) c.remove();
}

function renderFinalAnswer() {
    if (state.textBuf && state.textBuf.trim()) {
        dom.answerSection.classList.remove('empty');
        let html;
        try {
            html = marked.parse(state.textBuf);
        } catch (err) {
            console.error('marked failed', err);
            html = `<pre>${escapeHtml(state.textBuf)}</pre>`;
        }
        dom.answerSection.innerHTML = html;
        appendCodeCopyButtons(dom.answerSection);
    }
    removeCursor();
    if (state.stopReason === 'max_tokens') {
        const note = document.createElement('p');
        note.style.cssText = 'color:#888; font-size:12px; margin-top:12px; font-style:italic;';
        note.textContent = '…(响应被截断,达到 max_tokens 限制)';
        dom.answerSection.appendChild(note);
    }
}

function escapeHtml(s) {
    return String(s)
        .replace(/&/g, '&amp;')
        .replace(/</g, '&lt;')
        .replace(/>/g, '&gt;')
        .replace(/"/g, '&quot;')
        .replace(/'/g, '&#39;');
}

function appendCodeCopyButtons(root) {
    const pres = root.querySelectorAll('pre');
    pres.forEach((pre) => {
        if (pre.dataset.copyAttached === '1') return;
        pre.dataset.copyAttached = '1';
        const btn = document.createElement('button');
        btn.className = 'code-copy';
        btn.type = 'button';
        btn.textContent = '复制';
        btn.addEventListener('click', async () => {
            const code = pre.querySelector('code');
            const text = code ? code.textContent : pre.textContent;
            try {
                await navigator.clipboard.writeText(text);
            } catch (err) {
                const ta = document.createElement('textarea');
                ta.value = text;
                document.body.appendChild(ta);
                ta.select();
                try { document.execCommand('copy'); } catch (e) { /* ignore */ }
                document.body.removeChild(ta);
            }
            btn.textContent = '✓ 已复制';
            btn.classList.add('copied');
            setTimeout(() => {
                btn.textContent = '复制';
                btn.classList.remove('copied');
            }, 1500);
        });
        pre.appendChild(btn);
    });
}
```

async function send() {
    if (state.isStreaming) {
        if (state.abortController) state.abortController.abort();
        return;
    }
    const question = dom.questionInput.value.trim();
    if (!question) return;
    if (!state.apiKey) {
        showError('请先在顶部设置 API Key。');
        dom.apiKeyInput.focus();
        return;
    }
    showError(null);
    resetAnswer();
    state.question = question;
    state.abortController = new AbortController();
    setSendButtonState(true);

    try {
        await callApi(question);
        if (!state.textBuf && state.thinkingBuf) {
            state.textBuf = '（模型仅返回了思考过程,无最终答复。）';
            renderFinalAnswer();
        }
    } catch (err) {
        if (err.name === 'AbortError') {
            // user cancelled
        } else if (err.status) {
            const map = {
                401: 'API Key 无效或已过期。',
                403: '无权访问,请检查账户或区域。',
                404: '模型或接口路径错误。',
                429: '请求过于频繁,请稍后重试。',
            };
            const known = map[err.status];
            showError(known || `服务器错误 (${err.status}),请稍后重试。`);
        } else {
            showError(`网络错误: ${err.message || err}`);
        }
        removeCursor();
        renderFinalAnswer();
    } finally {
        setSendButtonState(false);
        state.abortController = null;
        updateSendEnabled();
    }
}

function clearAll() {
    if (state.isStreaming && state.abortController) {
        state.abortController.abort();
    }
    dom.questionInput.value = '';
    resetAnswer();
    showError(null);
    updateSendEnabled();
    dom.questionInput.focus();
}

dom.sendBtn.addEventListener('click', send);

dom.clearBtn.addEventListener('click', clearAll);

dom.questionInput.addEventListener('input', () => {
    dom.questionInput.style.height = 'auto';
    dom.questionInput.style.height = Math.min(dom.questionInput.scrollHeight, 200) + 'px';
    updateSendEnabled();
});

dom.questionInput.addEventListener('keydown', (e) => {
    if (e.ctrlKey && !e.shiftKey && !e.altKey && !e.metaKey && e.key === 'Enter') {
        e.preventDefault();
        send();
    } else if (e.key === 'Escape') {

});

updateSendEnabled();
